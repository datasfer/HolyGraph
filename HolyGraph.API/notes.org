** @surbgirqbot Mat 1 - 
/// Verse range
[HttpGet("{translation}/books/{book}/chapters/{chapter}/verses/{v1}/")]

 -----------------------------------
 Request: Verse Range enter
 URL: scripture/kjv-T/books/Matthew/chapters/1/verses/1? to=50
 MATCH(t:Translation) where t.id = 'kjv-T'
 MATCH(t)-[rt:CONTAINS]->(n:Book) where n.book = 'Matthew'
 match (n)<-[r:CONTAINS]->(ch:Chapter)-[:CONTAINS]->(v:Verse)
 where ch.num = 1 and v.verse >= 1 and v.verse <= 50
 return v, t.id as translation, n.title as bookTitle order by v.verseOrder
